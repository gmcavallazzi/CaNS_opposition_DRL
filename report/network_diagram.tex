\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, fit, calc, backgrounds}

\tikzset{
    % Network blocks (Actor/Critic)
    netblock/.style={rectangle, draw, very thick, fill=orange!10, text width=7em, text centered, rounded corners, minimum height=4em},
    % Action-related blocks
    actionblock/.style={rectangle, draw, fill=blue!10, text width=7em, text centered, rounded corners, minimum height=2.5em},
    actionwideblock/.style={rectangle, draw, fill=blue!10, text width=8em, text centered, rounded corners, minimum height=3em},
    % Action processing/correction blocks
    actionprocessblock/.style={rectangle, draw, fill=teal!15, text width=8em, text centered, rounded corners, minimum height=3em},
    % Observation-related blocks
    obsblock/.style={rectangle, draw, fill=orange!20, text width=7em, text centered, rounded corners, minimum height=2.5em},
    obsprocessblock/.style={rectangle, draw, fill=orange!20, text width=5em, text centered, rounded corners, minimum height=3em},
    % Reward/Q-value blocks
    rewardblock/.style={rectangle, draw, fill=yellow!10, text width=7em, text centered, rounded corners, minimum height=2.5em},
    % Critic input
    criticinputblock/.style={rectangle, draw, fill=pink!10, text width=7em, text centered, rounded corners, minimum height=2.5em},
    % Replay buffer
    bufferblock/.style={rectangle, draw, fill=gray!20, text width=8em, text centered, rounded corners, minimum height=3em},
    % Environment blocks
    envblock/.style={rectangle, draw, fill=green!10, text width=6em, text centered, rounded corners, minimum height=3em},
    % Loss block
    lossblock/.style={rectangle, draw, fill=red!10, text width=6em, text centered, rounded corners, minimum height=3em},
    % Arrows
    arrow/.style={-{Stealth[length=3mm]}, thick},
    bluearrow/.style={-{Stealth[length=3mm]}, thick, blue},
    orangearrow/.style={-{Stealth[length=3mm]}, thick, orange},
    purplearrow/.style={-{Stealth[length=3mm]}, thick, purple},
    grayarrow/.style={-{Stealth[length=3mm]}, thick, gray},
    gradient/.style={-{Stealth[length=3mm]}, thick, red},
}

\begin{document}
\begin{tikzpicture}[node distance=1.0cm, font=\small]

% Left: Environment
\node[envblock] (env) {CaNS\\Environment};

% Reward from environment (above env)
\node[rewardblock, above=1.2cm of env] (reward) {Reward\\$r_t = -\text{d}p/\text{d}x$\\(drag reduction)};

\node[obsblock, right=0.8cm of env] (obs) {Observations\\$[\mathbf{u}_t, \mathbf{w}_t]$\\(64×64 field)};

% Patch extraction
\node[obsprocessblock, right=0.8cm of obs] (patch) {Extract\\64 8×8 Patches};

% Previous action storage (above, centered in action memory region)
\node[actionblock, above=1.2cm of patch] (prevact) {Prev Action\\$\mathbf{a}_{t-1}$\\(64×64 field)};
\node[obsblock, right=0.8cm of patch] (obs3ch) {Observation\\$[\mathbf{u}, \mathbf{w}, \mathbf{a}_{t-1}]$\\(3×8×8)};

% Actor Network
\node[netblock, right=1.2cm of obs3ch] (actor) {\textbf{Actor CNN}\\Encoder-Decoder\\+ Similarity};
\node[actionblock, right=0.8cm of actor] (actraw) {Raw Actions\\$\mathbf{a}^{\text{raw}}_t$ $\tanh \in [-1,1]$};

% Zero-mean correction
\node[actionprocessblock, right=0.8cm of actraw] (zeromean) {Zero-Mean\\Correction\\(differentiable)};
\node[actionblock, right=0.6cm of zeromean] (actcorr) {$\mathbf{a}_t = \mathbf{a}^{\text{raw}}_t - \mu$\\$\mu = \text{mean}(\mathbf{a}^{\text{raw}}_t)$};

% Noise and zero-mean (combined block)
\node[actionprocessblock, right=0.8cm of actcorr] (noisemean) {Noise + Clip\\+ Zero-Mean\\(training)};
\node[actionblock, right=0.6cm of noisemean] (actfinal) {$\mathbf{a}^{\text{final}}_t$\\(exactly zero-mean)};

% Back to environment (complete the loop)
\node[envblock, right=0.8cm of actfinal] (execute) {Execute\\in CaNS};

% Critic Network (above the main flow)
\node[netblock, above=2.5cm of actcorr] (critic) {\textbf{Critic CNN}\\Circular Conv\\+ MLP};
\node[criticinputblock, left=0.8cm of critic] (criticinput) {Critic Input\\$[\mathbf{u}, \mathbf{w}, \mathbf{a}_{t-1}, \mathbf{a}_t]$\\(4×64×64)};
\node[rewardblock, right=0.8cm of critic] (qvalue) {Q-value\\$Q(\mathbf{s}_t, \mathbf{a}_t)$};

% Loss computation
\node[lossblock, right=0.8cm of qvalue] (actorloss) {Actor Loss\\$\mathcal{L} = -Q + \lambda_s \mathcal{L}_s$ $+ \lambda_g \mathcal{L}_g + \lambda_c \mathcal{L}_c$};

% Replay buffer (left of critic input)
\node[bufferblock, left=1.8cm of criticinput] (buffer) {Replay Buffer\\Store transitions};

% Draw arrows - Forward pass (horizontal flow)
\draw[arrow] (env) -- (obs);
\draw[arrow] (env) -- (reward);
\draw[arrow] (obs) -- (patch);
%\draw[arrow] (prevact) -| (patch);
\draw[arrow] (patch) -- (obs3ch);
\draw[arrow] (obs3ch) -- (actor);
\draw[arrow] (actor) -- (actraw);
\draw[arrow] (actraw) -- (zeromean);
\draw[arrow] (zeromean) -- (actcorr);
\draw[arrow] (actcorr) -- (noisemean);
\draw[arrow] (noisemean) -- (actfinal);

% Actions to environment (loop back below main flow)
\draw[arrow] (actfinal) -- (execute);
\draw[arrow] (execute.south) |- ($(execute.south)+(0,-1)$) -| (env.south);

% Actions to replay buffer (routed to avoid crossing blocks)
% actfinal in blue, approaches from below (offset from center to avoid overlap)
\draw[bluearrow] (actfinal.north) -- ++(0,1.6) -| ($(buffer.south)+(0.5,0)$);

% Previous action to extract patches
\draw[bluearrow] (prevact) -- (patch);

% obs3ch leaves from top (offset), goes up, right, up again to buffer from below (offset)
\draw[orangearrow] (obs3ch.north) -- ++(0,1.0) -| (buffer.south);

% reward approaches buffer from the left (west side)
\draw[grayarrow] (reward.north)  |- (buffer.west);

% Replay buffer to critic (training updates)
\draw[arrow, dashed] (buffer.east) -- (criticinput.west) node[midway, above, font=\small, align=left] {Training\\ batches};

% Critic branch - inputs from main flow
\draw[orangearrow] (obs3ch.north) -- ++(0,0.8) -| (criticinput.south);
\draw[purplearrow] (actcorr.north) -- ++(0,1.0) -| ($(criticinput.south)+(0.4,0)$);

% Critic forward pass (clear horizontal flow)
\draw[arrow] (criticinput) -- (critic);
\draw[arrow] (critic) -- (qvalue);
\draw[arrow] (qvalue) -- (actorloss);

% Gradient flow (red arrows - arched to avoid main flow, slightly off-center)
\draw[gradient] (actorloss.west) to[out=120, in=45] ($(critic.north)+(0.3,0)$);
\draw[gradient] (actorloss.south) to[out=270, in=90] ($(actcorr.north)+(0.2,0)$);
\draw[gradient] ($(actcorr.west)+(0,0.2)$) -- ($(zeromean.east)+(0,0.2)$);
\draw[gradient] ($(zeromean.west)+(0,0.2)$) -- ($(actraw.east)+(0,0.2)$);
\draw[gradient] ($(actraw.west)+(0,0.2)$) -- ($(actor.east)+(0,0.2)$);

% Nabla symbols on curved gradient arrows
\node[fill=white, inner sep=1pt, font=\large, text=red] at ($(critic.west)+(2.6,1.1)$) {$\nabla$};
\node[fill=white, inner sep=1pt, font=\large, text=red] at ($(actcorr.north)+(1.1,0.9)$) {$\nabla$};

% Store prev_action (loop back to top)
\draw[bluearrow] (actfinal) |- (prevact);

% Loss components annotation (larger font, positioned right)
\node[right=0.3cm of actorloss, text width=5cm, font=\small, align=left] {
    \textbf{Loss Components:}\\[2pt]
    $Q$: Q-value (from Critic)\\
    $\mathcal{L}_s$: Spatial smoothness\\
    $\mathcal{L}_g$: Global zero-mean\\
    $\mathcal{L}_c$: Consistency
};

\begin{scope}[on background layer]

\end{scope}
\node[below left=-0.2cm and -0.2cm of actionmemory.south east, fill=blue!5, font=\small\bfseries] {Action Memory};

\begin{scope}[on background layer]
\node[draw, fill=orange!5, thick, rounded corners, fit={(zeromean) (actcorr) (noisemean)}, inner sep=10pt] (zeroconstraint) {};
\end{scope}
\node[below left=-0.1cm and -0.35cm of zeroconstraint.south east, fill=orange!5, font=\small\bfseries] {Zero-Mean Constraint};

\end{tikzpicture}
